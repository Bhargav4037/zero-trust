package http.authz

default allow = false

# Decode JWT and extract roles
parsed_token := io.jwt.decode_verify(
  input.attributes.request.http.headers.authorization,
  {"secret": {"algorithm": "RS256", "cert": "<PEM PUBLIC KEY HERE>"}}
)

roles := parsed_token.payload.realm_access.roles

# Allow GET on /bookdetails (open to all)
allow {
  input.attributes.request.http.method == "GET"
  input.attributes.request.http.path == "/bookdetails"
}

# Allow GET on /getbooks or /getbookbytitle* (if any authenticated user)
allow {
  input.attributes.request.http.method == "GET"
  startswith(input.attributes.request.http.path, "/getbookbytitle") or
  input.attributes.request.http.path == "/getbooks"
  input.parsed_token.payload.sub != ""
}

# Allow POST to /addbook* if user has 'admin' role
allow {
  input.attributes.request.http.method == "POST"
  startswith(input.attributes.request.http.path, "/addbook")
  roles[_] == "admin"
}
